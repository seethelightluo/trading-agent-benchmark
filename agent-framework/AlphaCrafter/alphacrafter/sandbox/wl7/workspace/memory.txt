import json
from pathlib import Path
import numpy as np
from alphacrafter.sim.utils import register_hook, get_stock_daily_data, rebalance_to_weights

UNIVERSE = ["000300.SH", "SPX", "HSI", "N225", "SX5E", "000688.SH", "SOX", "NDX", "XAU", "COPPER", "WTI", "BTC", "ETH", "US10Y", "CN10Y"]
ENSEMBLE_PATH = Path(__file__).parent / "factors" / "factor_ensemble.json"
REBALANCE_DAYS = 10
MIN_WEIGHT, MAX_WEIGHT = 0.015, 0.16
_last_decision = None


def cs_rank(values, direction=1):
    valid = sorted((s, direction * float(v)) for s, v in values.items() if np.isfinite(v))
    out = {s: 0.5 for s in UNIVERSE}
    for i, (s, _) in enumerate(valid):
        out[s] = (i + 1.0) / max(len(valid), 1)
    return out


def bounded(raw):
    w = {s: max(float(raw.get(s, 0.01)), 1e-12) for s in UNIVERSE}
    fixed = set()
    for _ in range(30):
        free = [s for s in UNIVERSE if s not in fixed]
        scale = (1.0 - sum(w[s] for s in fixed)) / max(sum(w[s] for s in free), 1e-12)
        changed = False
        for s in free:
            w[s] *= scale
            if w[s] < MIN_WEIGHT:
                w[s] = MIN_WEIGHT; fixed.add(s); changed = True
            elif w[s] > MAX_WEIGHT:
                w[s] = MAX_WEIGHT; fixed.add(s); changed = True
        if not changed:
            break
    total = sum(w.values())
    return {s: w[s] / total for s in UNIVERSE}


def ensemble():
    try:
        obj = json.loads(ENSEMBLE_PATH.read_text(encoding="utf-8"))
        selected = [x for x in obj.get("selected_factors", []) if x.get("factor_id")][:10]
        weights = {x["factor_id"]: float(x.get("weight", 0.0)) for x in selected}
        directions = {x["factor_id"]: int(x.get("direction", 1)) for x in selected}
        return selected, weights, directions
    except Exception:
        return [], {}, {}


@register_hook
def cross_asset_strategy():
    global _last_decision
    selected, fw, directions = ensemble()
    if not selected:
        return
    data = {}
    for s in UNIVERSE:
        df = get_stock_daily_data(symbol=s, days=90)
        if df is None or len(df) < 25:
            continue
        df = df.sort_values("date").reset_index(drop=True)
        c = np.asarray(df["close"], dtype=float)
        h = np.asarray(df["high"], dtype=float)
        l = np.asarray(df["low"], dtype=float)
        r = c[1:] / np.maximum(c[:-1], 1e-12) - 1.0
        data[s] = (c, h, l, r, str(df.iloc[-1]["date"]))
    if len(data) < 12:
        return
    decision = max(v[4] for v in data.values())
    if _last_decision is not None:
        try:
            if (np.datetime64(decision) - np.datetime64(_last_decision)) / np.timedelta64(1, "D") < REBALANCE_DAYS:
                return
        except Exception:
            return

    factors = {fid: {} for fid in fw}
    invvol, ret5 = {}, {}
    for s, (c, h, l, r, _) in data.items():
        vol = max(float(np.std(r[-20:])), 0.008)
        clv = float((2*c[-1] - h[-1] - l[-1]) / max(h[-1] - l[-1], 1e-12))
        reversal5 = -float(np.mean(r[-5:]))
        reversal1 = -float(r[-1])
        momentum = float((c[-1] / max(c[-21], 1e-12) - 1.0) / (vol + 0.01))
        ret5[s] = float(c[-1] / max(c[-6], 1e-12) - 1.0)
        invvol[s] = 1.0 / vol
        values = {
            "miner_3_clv_1d": clv,
            "short_term_reversal_5d": reversal5,
            "miner_3_intraday_reversal_1d": reversal1,
            "miner_2_risk_adjusted_momentum_20d": momentum,
        }
        for fid, value in values.items():
            if fid in factors: factors[fid][s] = value
    med = float(np.median(list(ret5.values())))
    if "peer_median_leadlag_5d" in factors:
        factors["peer_median_leadlag_5d"] = {s: v - med for s, v in ret5.items()}

    ranks = {fid: cs_rank(factors.get(fid, {}), directions.get(fid, 1)) for fid in fw}
    score = {s: sum(fw[fid] * ranks[fid].get(s, 0.5) for fid in fw) for s in UNIVERSE}
    # Mild bull: retain signal exposure. In a confirmed weak SPX tape, rotate
    # toward defensive tradable benchmarks while remaining fully invested.
    if "SPX" in data and data["SPX"][0][-1] < data["SPX"][0][-21]:
        for s in ("XAU", "US10Y", "CN10Y"): score[s] += 0.10
        for s in ("BTC", "ETH", "WTI"): score[s] = max(0.01, score[s] - 0.06)
    avg_iv = float(np.mean(list(invvol.values())))
    raw = {s: max(score[s], 0.01) * (0.78 + 0.22 * invvol.get(s, avg_iv) / max(avg_iv, 1e-12)) for s in UNIVERSE}
    weights = bounded(raw)
    mean, sd = float(np.mean(list(score.values()))), max(float(np.std(list(score.values()))), 1e-12)
    forecast = {s: float(0.01 * (score[s] - mean) / sd) for s in UNIVERSE}
    rebalance_to_weights(weights, forecast_returns=forecast,
                         factor_ids=[x["factor_id"] for x in selected], horizon_days=10)
    _last_decision = decision
